十大排序算法

2025-03-25
  1. 冒泡排序 (Bubble Sort)
    原理：通过相邻元素之间的比较和交换，逐步将较大（或较小）的元素“冒泡”到数组的一端。
    时间复杂度：O(n²)
// 冒泡排序方法
    public static void bubbleSort(int[] arr) {
        int n = arr.length;
        // 外层循环控制排序的轮数
        for (int i = 0; i < n - 1; i++) {
            // 内层循环进行相邻元素的比较和交换
            for (int j = 0; j < n - 1 - i; j++) {
                // 如果当前元素大于下一个元素，则交换
                if (arr[j] > arr[j + 1]) {
                    // 交换元素
                    int temp = arr[j];
                    arr[j] = arr[j + 1];
                    arr[j + 1] = temp;
                }
            }
        }
    }

    public static void betterBubbleSort(int[] arr) {
        int n = arr.length;
        // 外层循环控制排序的轮数
        for (int i = 0; i < n - 1; i++) {
            // 标志位，记录本轮是否发生过交换
            boolean swapped = false;

            // 内层循环进行相邻元素的比较和交换
            for (int j = 0; j < n - 1 - i; j++) {
                // 如果当前元素大于下一个元素，则交换
                if (arr[j] > arr[j + 1]) {
                    // 交换元素
                    int temp = arr[j];
                    arr[j] = arr[j + 1];
                    arr[j + 1] = temp;
                    swapped = true; // 标记发生了交换
                }
            }

            // 如果本轮没有发生交换，说明数组已经有序，直接退出
            if (!swapped) {
                break;
            }
        }
    }
  1. 选择排序 (Selection Sort)
    原理：每次从未排序部分中选出最小（或最大）的元素，放到已排序部分的末尾。
    时间复杂度：O(n²)
// 选择排序方法
    public static void selectionSort(int[] arr) {
        int n = arr.length;
        // 外层循环控制已排序部分的边界
        for (int i = 0; i < n - 1; i++) {
            // 假设当前未排序部分的第一个元素是最小的
            int minIndex = i;

            // 内层循环从未排序部分中寻找最小元素的索引
            for (int j = i + 1; j < n; j++) {
                if (arr[j] < arr[minIndex]) {
                    minIndex = j; // 更新最小元素索引
                }
            }

            // 将找到的最小元素与未排序部分的第一个元素交换
            if (minIndex != i) {
                int temp = arr[i];
                arr[i] = arr[minIndex];
                arr[minIndex] = temp;
            }
        }
    }
  1. 插入排序 (Insertion Sort)
    原理：将数组分为已排序和未排序两部分，逐个将未排序的元素插入到已排序部分中的正确位置。
    时间复杂度：O(n²)
// 插入排序方法
    public static void insertionSort(int[] arr) {
        int n = arr.length;
        // 从第二个元素开始（索引 1），逐个插入到已排序部分
        for (int i = 1; i < n; i++) {
            int key = arr[i]; // 当前待插入的元素
            int j = i - 1;    // 已排序部分的最后一个元素索引

            // 将大于 key 的元素向后移动，为 key 腾出位置
            while (j >= 0 && arr[j] > key) {
                arr[j + 1] = arr[j]; // 元素后移
                j--;
            }
            // 插入 key 到正确位置
            arr[j + 1] = key;
        }
    }
  1. 希尔排序 (Shell Sort)
    原理：插入排序的改进版，通过设置间隔（步长）对元素分组排序，逐步缩小间隔直到完成排序。
    时间复杂度：O(n log n) 至 O(n²)，取决于步长序列
// 希尔排序方法
    public static void shellSort(int[] arr) {
        int n = arr.length;

        // 从初始增量（gap）开始，逐步缩小直到 1
        for (int gap = n / 2; gap > 0; gap /= 2) {
            // 对每个增量分组进行插入排序
            for (int i = gap; i < n; i++) {
                int key = arr[i]; // 当前待插入的元素
                int j = i;

                // 在当前组内进行插入排序，将 key 插入到正确位置
                while (j >= gap && arr[j - gap] > key) {
                    arr[j] = arr[j - gap]; // 元素后移
                    j -= gap;
                }
                arr[j] = key; // 插入 key
            }
        }
    }
  1. 归并排序 (Merge Sort)
    原理：采用分治法，将数组递归分成小块，排序后再合并。
    归并排序有两种实现方法，一种是自顶向下，一种是自底向上
    时间复杂度：O(n log n)
    自底向上版本：
// 自底向上归并排序入口
    public static void mergeSort(int[] arr) {
        if (arr == null || arr.length <= 1) {
            return;
        }
        int n = arr.length;

        // 从子数组大小 1 开始，逐步加倍合并
        for (int size = 1; size < n; size *= 2) {
            // 对每个子数组对进行合并
            for (int left = 0; left < n - size; left += size * 2) {
                int mid = left + size - 1;
                int right = Math.min(left + size * 2 - 1, n - 1); // 防止越界
                merge(arr, left, mid, right);
            }
        }
    }

    // 合并两个有序子数组
    private static void merge(int[] arr, int left, int mid, int right) {
        int n1 = mid - left + 1;
        int n2 = right - mid;
        int[] leftArray = new int[n1];
        int[] rightArray = new int[n2];

        // 复制数据到临时数组
        for (int i = 0; i < n1; i++) leftArray[i] = arr[left + i];
        for (int j = 0; j < n2; j++) rightArray[j] = arr[mid + 1 + j];

        // 合并
        int i = 0, j = 0, k = left;
        while (i < n1 && j < n2) {
            if (leftArray[i] <= rightArray[j]) { // 保证稳定性
                arr[k++] = leftArray[i++];
            } else {
                arr[k++] = rightArray[j++];
            }
        }
        while (i < n1) arr[k++] = leftArray[i++];
        while (j < n2) arr[k++] = rightArray[j++];
    }

自顶向下版本：

// 自顶向下归并排序入口
    public static void mergeSort(int[] arr) {
        if (arr == null || arr.length <= 1) {
            return;
        }
        sort(arr, 0, arr.length - 1);
    }

    // 递归分治
    private static void sort(int[] arr, int left, int right) {
        if (left < right) {
            int mid = left + (right - left) / 2;
            sort(arr, left, mid);        // 递归排序左半部分
            sort(arr, mid + 1, right);   // 递归排序右半部分
            merge(arr, left, mid, right); // 合并
        }
    }

    // 合并两个有序子数组
    private static void merge(int[] arr, int left, int mid, int right) {
        int n1 = mid - left + 1;
        int n2 = right - mid;
        int[] leftArray = new int[n1];
        int[] rightArray = new int[n2];

        // 复制数据到临时数组
        for (int i = 0; i < n1; i++) leftArray[i] = arr[left + i];
        for (int j = 0; j < n2; j++) rightArray[j] = arr[mid + 1 + j];

        // 合并
        int i = 0, j = 0, k = left;
        while (i < n1 && j < n2) {
            if (leftArray[i] <= rightArray[j]) { // 保证稳定性
                arr[k++] = leftArray[i++];
            } else {
                arr[k++] = rightArray[j++];
            }
        }
        while (i < n1) arr[k++] = leftArray[i++];
        while (j < n2) arr[k++] = rightArray[j++];
    }
  1. 快速排序 (Quick Sort)
    原理：选择一个基准（pivot），将数组分为小于和大于基准的两部分，递归排序。
    时间复杂度：平均 O(n log n)，最坏 O(n²)
// 快速排序入口方法
    public static void quickSort(int[] arr) {
        if (arr == null || arr.length <= 1) {
            return; // 数组为空或只有一个元素，无需排序
        }
        sort(arr, 0, arr.length - 1);
    }

    // 递归排序方法
    private static void sort(int[] arr, int low, int high) {
        if (low < high) {
            // 分区并获取基准位置
            int pivotIndex = partition(arr, low, high);
            // 递归排序左半部分
            sort(arr, low, pivotIndex - 1);
            // 递归排序右半部分
            sort(arr, pivotIndex + 1, high);
        }
    }

    // 分区方法（Lomuto 分区方案）
    private static int partition(int[] arr, int low, int high) {
        int pivot = arr[high]; // 选择最后一个元素作为基准
        int i = low - 1;       // i 指向小于基准的区域边界

        // 遍历从 low 到 high-1 的元素
        for (int j = low; j < high; j++) {
            // 如果当前元素小于等于基准，扩展小于区域
            if (arr[j] <= pivot) {
                i++;
                // 交换元素
                int temp = arr[i];
                arr[i] = arr[j];
                arr[j] = temp;
            }
        }

        // 将基准放到正确位置
        int temp = arr[i + 1];
        arr[i + 1] = arr[high];
        arr[high] = temp;

        return i + 1; // 返回基准最终位置
    }
  1. 堆排序 (Heap Sort)
    原理：利用堆数据结构（通常是最大堆或最小堆），反复调整堆并提取元素完成排序。
    时间复杂度：O(n log n)
// 堆排序入口方法
    public static void heapSort(int[] arr) {
        if (arr == null || arr.length <= 1) {
            return; // 数组为空或只有一个元素，无需排序
        }

        int n = arr.length;

        // 第一步：构建最大堆
        for (int i = n / 2 - 1; i >= 0; i--) {
            heapify(arr, n, i);
        }

        // 第二步：逐步将堆顶（最大值）移到末尾，并调整剩余部分为最大堆
        for (int i = n - 1; i > 0; i--) {
            // 将堆顶（最大值）与当前末尾交换
            swap(arr, 0, i);
            // 对剩余部分调整为最大堆
            heapify(arr, i, 0);
        }
    }

    // 调整堆的方法（最大堆）
    private static void heapify(int[] arr, int n, int i) {
        int largest = i;       // 假设当前节点为最大
        int left = 2 * i + 1;  // 左子节点
        int right = 2 * i + 2; // 右子节点

        // 如果左子节点存在且大于当前最大值
        if (left < n && arr[left] > arr[largest]) {
            largest = left;
        }

        // 如果右子节点存在且大于当前最大值
        if (right < n && arr[right] > arr[largest]) {
            largest = right;
        }

        // 如果最大值不是当前节点，交换并递归调整
        if (largest != i) {
            swap(arr, i, largest);
            heapify(arr, n, largest); // 递归调整受影响的子树
        }
    }

    // 交换数组中两个元素
    private static void swap(int[] arr, int i, int j) {
        int temp = arr[i];
        arr[i] = arr[j];
        arr[j] = temp;
    }
  1. 计数排序 (Counting Sort)
    原理：非比较排序，统计每个元素出现的次数，然后根据计数重建有序数组。
    时间复杂度：O(n + k)，k 是数据范围
// 计数排序方法
    public static void countingSort(int[] arr) {
        if (arr == null || arr.length <= 1) {
            return; // 数组为空或只有一个元素，无需排序
        }

        // 第一步：找到数组中的最大值，确定计数数组的大小
        int max = arr[0];
        for (int i = 1; i < arr.length; i++) {
            if (arr[i] > max) {
                max = arr[i];
            }
        }

        // 第二步：创建计数数组并统计每个元素出现的次数
        int[] count = new int[max + 1]; // 索引从 0 到 max
        for (int i = 0; i < arr.length; i++) {
            count[arr[i]]++;
        }

        // 第三步：将计数数组转换为累加数组（表示小于等于某个值的元素个数）
        for (int i = 1; i <= max; i++) {
            count[i] += count[i - 1];
        }

        // 第四步：从原始数组逆序遍历，构建排序后的结果
        int[] output = new int[arr.length];
        for (int i = arr.length - 1; i >= 0; i--) {
            output[count[arr[i]] - 1] = arr[i]; // 放入正确位置
            count[arr[i]]--;                    // 更新计数
        }

        // 第五步：将结果复制回原数组
        for (int i = 0; i < arr.length; i++) {
            arr[i] = output[i];
        }
    }
  1. 桶排序 (Bucket Sort)
    原理：将元素分配到若干个桶中，对每个桶分别排序，最后合并结果。
    时间复杂度：平均 O(n + k)，k 是桶的数量
// 桶排序方法
    public static void bucketSort(float[] arr) {
        if (arr == null || arr.length <= 1) {
            return; // 数组为空或只有一个元素，无需排序
        }

        int n = arr.length;

        // 第一步：创建桶（这里用 ArrayList 数组表示）
        ArrayList<Float>[] buckets = new ArrayList[n];
        for (int i = 0; i < n; i++) {
            buckets[i] = new ArrayList<>();
        }

        // 第二步：将元素分配到对应的桶中
        for (int i = 0; i < n; i++) {
            int bucketIndex = (int) (n * arr[i]); // 假设 arr[i] 在 [0, 1) 范围内
            buckets[bucketIndex].add(arr[i]);
        }

        // 第三步：对每个桶内的元素进行排序（使用 Collections.sort 或插入排序）
        for (int i = 0; i < n; i++) {
            Collections.sort(buckets[i]); // 使用 Java 自带的稳定排序
        }

        // 第四步：将所有桶的元素依次放回原数组
        int index = 0;
        for (int i = 0; i < n; i++) {
            for (float num : buckets[i]) {
                arr[index++] = num;
            }
        }
    }
  1. 基数排序 (Radix Sort)
    原理：非比较排序，按位（个位、十位等）对数字进行多次分配和收集，最终排序。
    时间复杂度：O(n * k)，k 是最大位数
// 基数排序入口方法
   public static void radixSort(int[] arr) {
       if (arr == null || arr.length <= 1) {
           return; // 数组为空或只有一个元素，无需排序
       }

       // 找到最大值以确定最大位数
       int max = arr[0];
       for (int i = 1; i < arr.length; i++) {
           if (arr[i] > max) {
               max = arr[i];
           }
       }

       // 按位进行计数排序，从最低位到最高位
       for (int exp = 1; max / exp > 0; exp *= 10) {
           countingSortByDigit(arr, exp);
       }
   }

   // 按某一位进行计数排序
   private static void countingSortByDigit(int[] arr, int exp) {
       int n = arr.length;
       int[] output = new int[n]; // 输出数组
       int[] count = new int[10]; // 计数数组（0-9）

       // 统计当前位上每个数字的出现次数
       for (int i = 0; i < n; i++) {
           int digit = (arr[i] / exp) % 10;
           count[digit]++;
       }

       // 转换为累加计数，表示每个数字的位置
       for (int i = 1; i < 10; i++) {
           count[i] += count[i - 1];
       }

       // 从右到左构建排序结果，保证稳定性
       for (int i = n - 1; i >= 0; i--) {
           int digit = (arr[i] / exp) % 10;
           output[count[digit] - 1] = arr[i];
           count[digit]--;
       }

       // 将结果复制回原数组
       for (int i = 0; i < n; i++) {
           arr[i] = output[i];
       }
   }

综合对比

算法 时间复杂度 空间复杂度 稳定性 原地排序 适用场景
冒泡排序 O(n²) O(1) 稳定 是 小规模、部分有序
选择排序 O(n²) O(1) 不稳定 是 小规模、交换代价高
插入排序 O(n²) O(1) 稳定 是 小规模、在线排序
希尔排序 O(n^1.3) O(1) 不稳定 是 中等规模
归并排序 O(n log n) O(n) 稳定 否 大规模、链表、稳定性
快速排序 O(n log n) O(log n) 不稳定 是 大规模、一般用途
堆排序 O(n log n) O(1) 不稳定 是 固定复杂度、优先队列
计数排序 O(n + k) O(n + k) 稳定 否 范围有限整数、稳定性
桶排序 O(n) O(n + k) 稳定 否 均匀分布数据
基数排序 O(d * (n + k)) O(n + k) 稳定 否 大规模整数、多条件排序
